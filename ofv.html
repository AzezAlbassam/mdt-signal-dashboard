//      the defaults below reproduce that chart; that this is his formula is NOT shown. Pre-registered for his
//      next chart in PROTOCOL-study15-addendum-2.md.
// The gamma levels below follow PROTOCOL-study15.md §4 and are untested against his.
//
// Line  = session VWAP of 1-minute hlc3 from 09:30 ET + "start delay" (weights: Volume, Equal, or 0DTE options flow).
// Bands = line ± 1σ, ± 2σ; σ = same-weight deviation of hlc3 around the line.
// Use it on SP:SPX (its volume is the constituents' total); CBOE:SPX has no volume, use Equal there.
// Gamma = 0DTE SPXW calls/puts at 5-point strikes around the open, last session on the chart only:
//         Black-Scholes IV per traded bar (r = q = 0), gamma at the current spot × cumulative session
//         volume × 100 × S² × 1%; calls +, puts −. Yellow = calls dominate, purple = puts dominate.
//         Needs OPRA data on your TradingView plan. Strikes are fixed at the open (± 5 × "strikes each side").
// Not yet compiled on TradingView: compile once and report any error.

// ── Black-Scholes, r = q = 0 ──
erf(float x) =>
    // Abramowitz & Stegun 7.1.26, |error| < 1.5e-7
    t = 1.0 / (1.0 + 0.3275911 * math.abs(x))
    y = 1.0 - (((((1.061405429 * t - 1.453152027) * t) + 1.421413741) * t - 0.284496736) * t + 0.254829592) * t * math.exp(-x * x)
    x >= 0 ? y : -y
ncdf(float x) => 0.5 * (1.0 + erf(x / math.sqrt(2.0)))
npdf(float x) => math.exp(-0.5 * x * x) / math.sqrt(2.0 * math.pi)
bsPrice(bool isCall, float S, float K, float T, float v) =>
    d1 = (math.log(S / K) + 0.5 * v * v * T) / (v * math.sqrt(T))
    d2 = d1 - v * math.sqrt(T)
    isCall ? S * ncdf(d1) - K * ncdf(d2) : K * ncdf(-d2) - S * ncdf(-d1)
bsGamma(float S, float K, float T, float v) =>
    d1 = (math.log(S / K) + 0.5 * v * v * T) / (v * math.sqrt(T))
    npdf(d1) / (S * v * math.sqrt(T))
impliedVol(bool isCall, float p, float S, float K, float T) =>
    float res = na
    intrinsic = math.max(0.0, isCall ? S - K : K - S)
    if not na(p) and p >= intrinsic + 0.01 and p >= bsPrice(isCall, S, K, T, 0.01) and p <= bsPrice(isCall, S, K, T, 3.0)
        lo = 0.01
        hi = 3.0
        for it = 1 to 40
            mid = 0.5 * (lo + hi)
            if bsPrice(isCall, S, K, T, mid) < p
                lo := mid
            else
                hi := mid
        res := 0.5 * (lo + hi)
    res

dayKey(int t) => year(t, TZ) * 10000 + month(t, TZ) * 100 + dayofmonth(t, TZ)
isLastDay = dayKey(time) == dayKey(last_bar_time)

    ymd = str.tostring(year(time, TZ) % 100, "00") + str.tostring(month(time, TZ), "00") + str.tostring(dayofmonth(time, TZ), "00")
    array.clear(tickers)
    array.clear(strikes)
    for i = 0 to nK - 1

        array.push(strikes, K)
        array.push(tickers, "OPRA:SPXW" + ymd + "C" + str.tostring(K) + ".0")
        array.push(tickers, "OPRA:SPXW" + ymd + "P" + str.tostring(K) + ".0")
    array.fill(ivs, na)
    array.fill(cvs, 0.0)
closeT = timestamp(TZ, year(time, TZ), month(time, TZ), dayofmonth(time, TZ), 16, 0)

        [oc, ov] = request.security(array.get(tickers, j), timeframe.period, [close, volume], gaps = barmerge.gaps_on, ignore_invalid_symbol = true)

                array.set(ivs, j, ivNew)   // carried forward between trades
        ivNow = array.get(ivs, j)

        if not na(ivNow) and cv > 0

            array.set(expo, jk, array.get(expo, jk) + e)
